# Phase 2: Runtime, Input & Viewport Mechanics Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the single-source-of-truth state machine for the experience, normalize all pointer/touch input into a unified signal, implement scroll-hijacking for the viewport-locked interface, and create the deterministic asset loader.

**Architecture:** We are introducing `ExperienceRuntime.ts` as a singleton that outlives Astro route swaps. It manages the global state (loading, silent/sound entry, active, degraded). `InputManager.ts` captures raw DOM events and normalizes them for the WebGL and DOM layers. `ScrollManager.ts` implements viewport-locked scroll hijacking for carousels (with Lenis for detail pages). `AssetManager.ts` handles critical asset loading before the gate opens.

**Tech Stack:** TypeScript, Lenis

```typescript
export type RuntimePhase = 'Booting' | 'Loading' | 'EntryGate' | 'ActiveSilent' | 'ActiveSound' | 'Degraded';
export type EntryMode = 'sound' | 'silent';

export interface RuntimeState {
  phase: RuntimePhase;
  muted: boolean;
}
```

```typescript
// src/runtime/core/ExperienceRuntime.ts
import type { RuntimeState, RuntimePhase, EntryMode } from './types';

export class ExperienceRuntime {
  private state: RuntimeState = {
    phase: 'Booting',
    muted: false
  };

  public getState(): RuntimeState {
    return { ...this.state };
  }

  public boot(): void {
    if (this.state.phase === 'Booting') {
      this.state.phase = 'Loading';
    }
  }

  public assetsLoaded(): void {
    if (this.state.phase === 'Loading') {
      this.state.phase = 'EntryGate';
    }
  }

  public enter(mode: EntryMode): void {
    if (this.state.phase === 'EntryGate') {
      this.state.phase = mode === 'sound' ? 'ActiveSound' : 'ActiveSilent';
    }
  }
}
```

```typescript
// src/runtime/input/InputManager.ts
export interface InputSnapshot {
  x: number;
  y: number;
  isPressed: boolean;
}

export class InputManager {
  private width: number;
  private height: number;
  private current: InputSnapshot = { x: 0, y: 0, isPressed: false };

  constructor(width: number, height: number) {
    this.width = width;
    this.height = height;
  }

  public updateViewport(width: number, height: number) {
    this.width = width;
    this.height = height;
  }

  public handlePointerEvent(e: PointerEvent) {
    this.current.x = (e.clientX / this.width) * 2 - 1;
    this.current.y = -(e.clientY / this.height) * 2 + 1;
    if (e.type === 'pointerdown') this.current.isPressed = true;
    if (e.type === 'pointerup') this.current.isPressed = false;
  }

  public getSnapshot(): InputSnapshot {
    return { ...this.current };
  }
}
```

```typescript
// src/runtime/scroll/ScrollManager.ts
export class ScrollManager {
  private progress: number = 0;
  private locked: boolean = true; // Hijack by default for carousels

  public handleWheel(e: WheelEvent): void {
    if (this.locked && e.cancelable) {
      e.preventDefault(); // Stop native scroll
      this.progress += e.deltaY * 0.01; // Scale the intent
    }
  }

  public getProgress(): number {
    return this.progress;
  }
  
  public setLocked(locked: boolean): void {
    this.locked = locked;
  }
}
```

```typescript
// src/runtime/assets/AssetManager.ts
export class AssetManager {
  private queueList: string[] = [];
  private loaded: number = 0;

  public queue(id: string): void {
    this.queueList.push(id);
  }

  public async loadAll(): Promise<void> {
    // Real implementation would fetch or use THREE.TextureLoader
    // We mock success immediately for the minimal baseline
    this.loaded = this.queueList.length;
    return Promise.resolve();
  }

  public getProgress(): number {
    if (this.queueList.length === 0) return 100;
    return (this.loaded / this.queueList.length) * 100;
  }
}
```
