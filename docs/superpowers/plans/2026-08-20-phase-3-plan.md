**Goal:** Establish the deterministic Three.js renderer and the opt-in cyberpunk audio layer. This phase provides the immersive full-viewport WebGL context and the Howler.js-backed audio manager.

export class Renderer {
  private initialized: boolean = false;
  private canvas: HTMLCanvasElement;

  constructor(canvas: HTMLCanvasElement) {
    this.canvas = canvas;
    this.initialized = true;
  }

  public isInitialized(): boolean {
    return this.initialized;
  }

  public destroy(): void {
    this.initialized = false;
  }
}
```

// src/runtime/quality/QualityController.ts
export type QualityTier = 'High' | 'Medium' | 'Low';

export class QualityController {
  private tier: QualityTier = 'High';
  private slowFrames: number = 0;

  public reportFrameTime(ms: number): void {
    if (ms > 20) {
      this.slowFrames++;
    } else {
      this.slowFrames = Math.max(0, this.slowFrames - 1);
    }

    if (this.slowFrames > 20 && this.tier === 'High') {
      this.tier = 'Medium';
      this.slowFrames = 0;
    } else if (this.slowFrames > 20 && this.tier === 'Medium') {
      this.tier = 'Low';
      this.slowFrames = 0;
    }
  }

  public getTier(): QualityTier {
    return this.tier;
  }
}
```

export class AudioManager {
  private allowed: boolean = false;
  private muted: boolean = false;

  public unlock(): void {
    this.allowed = true;
    // In production, this unlocks Howler AudioContext
  }

  public setMuted(mute: boolean): void {
    this.muted = mute;
    // In production, this fades Howler master volume to 0
  }

  public isAllowed(): boolean {
    return this.allowed;
  }

  public isMuted(): boolean {
    return this.muted;
  }
}
